@st.cache_data(ttl=10)
def get_current_price(symbol):
    try:
        url = f"https://fapi.binance.com/fapi/v1/ticker/price?symbol={symbol}"
        response = requests.get(url, timeout=2)
        if response.status_code == 200:
            return float(response.json()["price"])
    except Exception:
        pass
    return 0.0

@st.cache_data(ttl=10)
def fetch_live_data(symbol, interval):
    url = f"https://fapi.binance.com/fapi/v1/klines?symbol={symbol}&interval={interval}&limit=100"
    response = requests.get(url, timeout=3)
    response.raise_for_status()
    data = response.json()
    df = pd.DataFrame(data, columns=[
        "time", "open", "high", "low", "close", "volume",
        "close_time", "qav", "num_trades", "taker_base_vol", "taker_quote_vol", "ignore"
    ])
    for col in ["open", "high", "low", "close", "volume"]:
        df[col] = pd.to_numeric(df[col], errors="coerce")
    df["time"] = pd.to_datetime(df["time"], unit="ms")
    return df

# STABLE SIGNAL ENGINE WITH TIMESTAMP EXTRACTION
# ============================================================

def get_signal_and_time(df):
    if df is None or len(df) < 25:
        return "CONSOLIDATING", None
    
    rsi = ta.rsi(df["close"], length=14)
    ema9 = ta.ema(df["close"], length=9)
    ema21 = ta.ema(df["close"], length=21)
    
    idx = -2 if len(df) > 26 else -1
    
    current_rsi = rsi.iloc[idx]
    current_ema9 = ema9.iloc[idx]
    current_ema21 = ema21.iloc[idx]
    candle_time = df["time"].iloc[idx]
    
    if any(pd.isna(v) for v in [current_rsi, current_ema9, current_ema21]):
        return "CONSOLIDATING", None
        
    avg_volume = df["volume"].rolling(20).mean().iloc[idx]
    whale_volume = not pd.isna(avg_volume) and df["volume"].iloc[idx] > avg_volume * 1.25
    price = df["close"].iloc[idx]
    recent_range = (df["high"].tail(8).max() - df["low"].tail(8).min()) / price
    is_consolidating = recent_range < 0.0030

    if is_consolidating:
        return "CONSOLIDATING", None
    elif current_ema9 > current_ema21 and current_rsi > 50 and whale_volume:
        return "BUY NOW", candle_time
    elif current_ema9 > current_ema21 and current_rsi > 46:
        return "UPCOMING UPTREND", candle_time
    elif current_rsi >= 68:
        return "PREPARE TO EXIT TP", candle_time
    elif current_ema9 < current_ema21 and current_rsi < 46 and whale_volume:
        return "EXIT NOW", candle_time
    elif current_ema9 < current_ema21 and current_rsi < 50:
        return "UPCOMING DOWNTREND", candle_time
    else:
        return "CONSOLIDATING", None

# HISTORICAL ENGINE
# ============================================================

def fetch_historical(symbol, timeframe, start_dt, end_dt):
    interval_ms = INTERVAL_MS[timeframe]
    start_ms = int(start_dt.timestamp() * 1000)
    end_ms = int(end_dt.timestamp() * 1000)
    rows = []
    current_ms = start_ms

    while current_ms < end_ms:
        url = f"https://fapi.binance.com/fapi/v1/klines?symbol={symbol}&interval={timeframe}&startTime={current_ms}&endTime={end_ms}&limit=1000"
        response = requests.get(url, timeout=5)
        response.raise_for_status()
        batch = response.json()
        if not batch:
            break
        rows.extend(batch)
        last_time = batch[-1][0]
        next_time = last_time + interval_ms
        if next_time <= current_ms:
            break
        current_ms = next_time
        time.sleep(0.01)

    if not rows:
        return pd.DataFrame()

    df = pd.DataFrame(rows, columns=[
        "time", "open", "high", "low", "close", "volume",
        "close_time", "qav", "num_trades", "taker_base_vol", "taker_quote_vol", "ignore"
    ])
    for col in ["open", "high", "low", "close", "volume"]:
        df[col] = pd.to_numeric(df[col], errors="coerce")
    df["time"] = pd.to_datetime(df["time"], unit="ms")
    return df.drop_duplicates(subset=["time"]).sort_values("time").reset_index(drop=True)

def replay_signals(df, symbol, timeframe, requested_start, requested_end):
    if df.empty:
        return pd.DataFrame()
    results = []
    previous_signal = None
    for index in range(25, len(df)):
        current_df = df.iloc[:index + 1]
        signal, candle_time = get_signal_and_time(current_df)
        if candle_time is None or candle_time < requested_start:
            previous_signal = signal
            continue
        if candle_time >= requested_end:
            break
        if signal != previous_signal and signal != "CONSOLIDATING":
            results.append({
                "time": candle_time, "symbol": symbol, "timeframe": timeframe,
                "signal": signal, "price": float(df.iloc[index]["close"])
            })
        previous_signal = signal
    if not results:
        return pd.DataFrame()
    return pd.DataFrame(results)

def run_historical_analysis(symbols, timeframe, start_dt, end_dt):
    all_results = []
    total_symbols = len(symbols)
    progress = st.progress(0, text="Preparing historical analysis...")
    for index, symbol in enumerate(symbols):
        progress.progress((index + 1) / total_symbols, text=f"Analyzing {symbol} ({index + 1}/{total_symbols})")
        warmup = {"1m": timedelta(hours=2), "3m": timedelta(hours=4), "5m": timedelta(hours=8), "1h": timedelta(days=2), "4h": timedelta(days=8)}[timeframe]
        try:
            df = fetch_historical(symbol, timeframe, start_dt - warmup, end_dt)
            result = replay_signals(df, symbol, timeframe, start_dt, end_dt)
            if not result.empty:
                all_results.append(result)
        except Exception:
            continue
    progress.empty()
    if not all_results:
        return pd.DataFrame()
    return pd.concat(all_results, ignore_index=True).sort_values("time").reset_index(drop=True)

def update_trade_tracker(symbol, timeframe, signal, price, candle_time):
    active = st.session_state.active_trades
    history = st.session_state.trade_history
    # Gumamit ng eksaktong candle_time kung mayroon, kung wala ay current time
    timestamp_to_use = candle_time if candle_time is not None else datetime.now()

    if signal == "BUY NOW" and symbol not in active:
        active[symbol] = {
            "entry_price": price, 
            "entry_time": timestamp_to_use.strftime("%Y-%m-%d %H:%M:%S"), 
            "tf": timeframe
        }
    elif signal in ["EXIT NOW", "PREPARE TO EXIT TP"] and symbol in active:
        entry_price = active[symbol]["entry_price"]
        entry_time = active[symbol]["entry_time"]
        saved_tf = active[symbol]["tf"]
        pnl_pct = ((price - entry_price) / entry_price) * 100
        result = "WIN" if pnl_pct > 0 else "LOSS"
        
        history.append({
            "symbol": symbol, 
            "tf": saved_tf, 
            "signal": "BUY TO EXIT",
            "entry_price": entry_price, 
            "exit_price": price, 
            "entry_time": entry_time,
            "exit_time": timestamp_to_use.strftime("%Y-%m-%d %H:%M:%S"),
            "pnl": pnl_pct, 
            "result": result
        })
        del active[symbol]

def get_badge_class(signal):
    if signal == "BUY NOW": return "buy"
    if signal == "EXIT NOW": return "exit"
    if signal == "PREPARE TO EXIT TP": return "tp"
    if signal == "UPCOMING UPTREND": return "up"
    if signal == "UPCOMING DOWNTREND": return "down"
    return "neutral"
